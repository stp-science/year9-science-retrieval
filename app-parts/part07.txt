  function gameCard(g,i){
    const unlocked=isUnlocked(g); const icons=['🩸','♥','⚡']; const desc=['Sort transport statements to the correct blood component.','Build the correct route through the heart and circulation.','Classify transport examples as diffusion, osmosis or active transport.'];
    return `<article class="game-card ${unlocked?'':'locked'}"><div class="game-icon">${icons[i]}</div><div class="unlock-tag">${unlocked?'Unlocked':'Locked'}</div><h3>${g.name}</h3><p>${desc[i]}</p><p><strong>Needs:</strong> ${g.req.map(k=>`${modules[k].short} ${mastery(k)}%`).join(' + ')}</p><button class="primary-btn" data-game="${g.id}" ${unlocked?'':'disabled'} type="button">${unlocked?'Play':'Keep revising'}</button></article>`;
  }

  const routeSteps=['Vena cava','Right atrium','Right ventricle','Pulmonary artery','Lungs','Pulmonary vein','Left atrium','Left ventricle','Aorta','Body'];
  function gameRoute(){
    const host=document.getElementById('game-host');let idx=0,errors=0;
    const draw=()=>{if(idx>=routeSteps.length){host.innerHTML=`<div class="game-stage"><h3>Route Rush complete</h3><p class="game-score">Route built with ${errors} error${errors===1?'':'s'}.</p><button class="secondary-btn" id="replay-game">Play again</button></div>`;document.getElementById('replay-game').onclick=gameRoute;if(errors<=2)addXP(15);return;}
