  const state = loadState();
  let activeModule = state.activeModule || 'blood';
  let activeView = 'learn';
  let activeLesson = 0;

  const panel = document.getElementById('content-panel');
  const moduleButtons = document.getElementById('module-buttons');
  const assessmentButton = document.getElementById('assessment-button');

  function loadState(){
    try { return JSON.parse(localStorage.getItem(STORAGE_KEY)) || {correct:{}, xp:0}; }
    catch { return {correct:{}, xp:0}; }
  }
  function saveState(){
    state.activeModule = activeModule;
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  }
  function addXP(amount){ state.xp = Math.max(0,(state.xp||0)+amount); saveState(); renderChrome(); }

  function renderChrome(){
    document.getElementById('xp-total').textContent = state.xp || 0;
    moduleButtons.innerHTML = Object.entries(modules).map(([key,m])=>{
      const qCount=questions[key]?.length || 0;
      const wCount=window.writtenQuestions?.[key]?.length || 0;
      const detail=activeView==='check' ? `${qCount} multiple choice • ${wCount} written` : m.subtitle;
      return `
      <button class="module-button ${activeModule===key && activeView!=='assessment'?'active':''}" data-module="${key}" type="button" style="--module-colour:${m.colour};--module-soft:${m.soft}">
        <span class="module-icon">${m.icon}</span><span><strong>${m.number} ${m.name}</strong><small>${detail}</small></span>
      </button>`;
    }).join('');
    moduleButtons.querySelectorAll('[data-module]').forEach(btn=>btn.addEventListener('click',()=>{
      activeModule=btn.dataset.module; activeLesson=0;
      if(activeView!=='check') { activeView='learn'; setActiveTab('learn'); }
      else setActiveTab('check');
      saveState(); render();
    }));
    assessmentButton.classList.toggle('active',activeView==='assessment');
    renderProgress();
    renderGamesBadge();
  }

  function mastery(key){
    const qs = questions[key] || [];
    const written = window.writtenQuestions?.[key] || [];
    const mcqHits = qs.filter((_,i)=>state.correct?.[`${key}-${i}`]).length;
    const writtenHits = written.filter((_,i)=>state.writtenCorrect?.[`${key}-written-${i}`]).length;
    const total = qs.length + written.length;
    return total ? Math.round(((mcqHits + writtenHits) / total) * 100) : 0;
  }
  function renderProgress(){
    const target=document.getElementById('dashboard-progress');
    target.innerHTML=Object.entries(modules).map(([key,m])=>{
      const pct=mastery(key);
      return `<div class="progress-item"><div class="progress-head"><span>${m.short}</span><strong>${pct}%</strong></div><div class="progress-track"><span style="width:${pct}%"></span></div></div>`;
    }).join('');
  }

    {id:'bloodblitz', name:'Blood Blitz', req:['blood','heart']},

  ];

  function renderGamesBadge(){document.getElementById('games-badge').textContent=`${unlocks.filter(isUnlocked).length}/3`;}

    document.querySelectorAll('.mode-tab').forEach(b=>b.classList.toggle('active',b.dataset.view===view));

    activeView=btn.dataset.view; setActiveTab(activeView); renderChrome(); render();
