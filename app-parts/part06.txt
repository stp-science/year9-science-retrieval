    if(activeView==='learn') renderLearn();
    else if(activeView==='check') renderCheck();
    else if(activeView==='games') renderGames();
    window.scrollTo({top:Math.max(0,document.querySelector('.mode-tabs').offsetTop-90),behavior:'smooth'});
  }

  function renderLearn(){
    const m=modules[activeModule];
    panel.style.setProperty('--accent',m.colour);
    activeLesson=Math.min(activeLesson,m.lessons.length-1);
    panel.innerHTML=`
      <p class="section-kicker">Section ${m.number}</p><h2>${m.name}</h2><p class="lead">${m.intro}</p>
      <div class="vocab-strip">${m.vocab.map(v=>`<span>${v}</span>`).join('')}</div>
      <div class="lesson-nav">${m.lessons.map((l,i)=>`<button class="lesson-chip ${i===activeLesson?'active':''}" data-lesson="${i}" type="button">${i+1}. ${l.title}</button>`).join('')}</div>
      <article class="lesson-body">${m.lessons[activeLesson].body()}</article>`;
    panel.querySelectorAll('[data-lesson]').forEach(btn=>btn.addEventListener('click',()=>{activeLesson=Number(btn.dataset.lesson);renderLearn();}));
  }

  function renderCheck(){
    const m=modules[activeModule], qs=questions[activeModule], written=window.writtenQuestions?.[activeModule]||[];
    panel.style.setProperty('--accent',m.colour);
    const score=qs.filter((_,i)=>state.correct?.[`${activeModule}-${i}`]).length;
    const writtenDone=written.filter((_,i)=>state.writtenChecked?.[`${activeModule}-written-${i}`]).length;
    panel.innerHTML=`
      <div class="quiz-head"><div><p class="section-kicker">Practice Questions • ${m.name}</p><h2>Multiple-choice + written practice</h2><p class="lead">Every course section has both question types. Complete the multiple-choice questions, then write your own answers to the short-answer questions before checking the key ideas and model answer.</p></div><div class="score-pill">MCQ ${score}/${qs.length} • Written ${writtenDone}/${written.length}</div></div>
      <h3>Multiple-choice practice</h3>
      ${qs.map((q,i)=>renderQuestion(q,i)).join('')}
      <h3 style="margin-top:28px">Written practice</h3>
      <p class="lead">Write in full sentences. Focus on the command word and use precise scientific vocabulary.</p>
      ${written.map((q,i)=>renderWrittenQuestion(q,i)).join('')}
      <button class="reset-btn" id="reset-section" type="button">Reset this section</button>`;
    panel.querySelectorAll('.option').forEach(btn=>btn.addEventListener('click',handleAnswer));
    panel.querySelectorAll('[data-written-input]').forEach(area=>area.addEventListener('input',e=>{
      if(!state.writtenAnswers) state.writtenAnswers={};
      state.writtenAnswers[e.currentTarget.dataset.writtenInput]=e.currentTarget.value;
      saveState();
    }));
    panel.querySelectorAll('[data-check-written]').forEach(btn=>btn.addEventListener('click',()=>checkWritten(Number(btn.dataset.checkWritten))));
    document.getElementById('reset-section').addEventListener('click',()=>{
      qs.forEach((_,i)=>delete state.correct[`${activeModule}-${i}`]);
      written.forEach((_,i)=>{
        const key=`${activeModule}-written-${i}`;
        if(state.writtenAnswers) delete state.writtenAnswers[key];
        if(state.writtenChecked) delete state.writtenChecked[key];
      });
      saveState();renderChrome();renderCheck();
    });
  }

  function renderQuestion(q,i){
    const key=`${activeModule}-${i}`; const done=!!state.correct?.[key];
    return `<div class="question-card" data-qindex="${i}"><div class="question-num">Multiple choice ${i+1} of ${questions[activeModule].length}</div><h3>${q.q}</h3><div class="options">${q.options.map((o,j)=>`<button class="option ${done&&j===q.a?'correct':''}" data-answer="${j}" ${done?'disabled':''} type="button">${String.fromCharCode(65+j)}. ${o}</button>`).join('')}</div>${done?`<div class="feedback good"><strong>Correct.</strong> ${q.explain}</div>`:''}</div>`;
  }

  function handleAnswer(e){
    const card=e.currentTarget.closest('.question-card'); const i=Number(card.dataset.qindex); const q=questions[activeModule][i]; const selected=Number(e.currentTarget.dataset.answer); const key=`${activeModule}-${i}`;
    card.querySelectorAll('.option').forEach((b,j)=>{b.disabled=true;if(j===q.a)b.classList.add('correct');if(j===selected&&j!==q.a)b.classList.add('wrong');});
    const fb=document.createElement('div');
    if(selected===q.a){
      if(!state.correct[key]) addXP(10); state.correct[key]=true; saveState();
      fb.className='feedback good';fb.innerHTML=`<strong>Correct.</strong> ${q.explain}`;
    }else{
      fb.className='feedback bad';fb.innerHTML=`<strong>Not yet.</strong> ${q.explain}<br><span>You can retry this question by reopening Practice Questions.</span>`;
      setTimeout(()=>{card.querySelectorAll('.option').forEach(b=>{b.disabled=false;b.classList.remove('correct','wrong')});fb.remove();},1800);
    }
    card.appendChild(fb); renderChrome();
  }

  function renderWrittenQuestion(q,i){
    const key=`${activeModule}-written-${i}`;
    const answer=state.writtenAnswers?.[key] || '';
    const checked=!!state.writtenChecked?.[key];
    return `<div class="question-card written-card" data-written-card="${i}">
      <div class="question-num">${q.command} • Written question ${i+1} of ${(window.writtenQuestions?.[activeModule]||[]).length}</div>
      <h3>${q.q}</h3>
      <textarea class="writing-area" style="min-height:125px" data-written-input="${key}" placeholder="Write your answer here before checking...">${escapeHtml(answer)}</textarea>
      <div style="display:flex;gap:8px;flex-wrap:wrap;margin-top:10px"><button class="primary-btn" data-check-written="${i}" type="button">Check your answer</button></div>
      <div id="written-feedback-${i}">${checked?writtenFeedback(q):''}</div>
    </div>`;
  }

  function escapeHtml(value){
    return String(value).replace(/[&<>'"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c]));
  }

  function writtenFeedback(q){
    return `<div class="assessment-box"><strong>Compare your answer with these key ideas</strong><ul>${q.keyPoints.map(x=>`<li>${x}</li>`).join('')}</ul></div>
      <details class="model-answer"><summary>Show model answer</summary><p>${q.model}</p><p><strong>Next step:</strong> close the model and improve your answer in your own words.</p></details>`;
  }

  function checkWritten(i){
    const key=`${activeModule}-written-${i}`;
    const area=panel.querySelector(`[data-written-input="${key}"]`);
    const target=document.getElementById(`written-feedback-${i}`);
    if(!area || area.value.trim().length<15){
      target.innerHTML='<div class="feedback bad"><strong>Write an answer first.</strong> Aim for at least one complete scientific sentence before checking.</div>';
      return;
    }
    if(!state.writtenChecked) state.writtenChecked={};
    if(!state.writtenChecked[key]) addXP(5);
    state.writtenChecked[key]=true;
    saveState();
    target.innerHTML=writtenFeedback(window.writtenQuestions[activeModule][i]);
    renderChrome();
  }
