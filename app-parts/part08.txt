
  const themeBtn=document.getElementById('theme-toggle');
  const savedTheme=localStorage.getItem(THEME_KEY);
  if(savedTheme==='dark'){document.body.classList.add('dark');themeBtn.textContent='☀';themeBtn.setAttribute('aria-label','Use light theme');}
  themeBtn.addEventListener('click',()=>{document.body.classList.toggle('dark');const dark=document.body.classList.contains('dark');localStorage.setItem(THEME_KEY,dark?'dark':'light');themeBtn.textContent=dark?'☀':'☾';themeBtn.setAttribute('aria-label',dark?'Use light theme':'Use dark theme');});

  const zoomStyle=document.createElement('style');
  zoomStyle.id='image-zoom-styles';
  zoomStyle.textContent=`
    #content-panel img.zoomable-image{cursor:zoom-in;transition:transform .15s ease,box-shadow .15s ease}
    #content-panel img.zoomable-image:hover{transform:scale(1.01);box-shadow:0 8px 22px rgba(0,0,0,.16)}
    #content-panel img.zoomable-image:focus{outline:3px solid #2d6fb5;outline-offset:3px}
    body.image-zoom-open{overflow:hidden}
    .image-zoom-overlay{position:fixed;inset:0;z-index:9999;background:rgba(3,12,22,.92);display:none;align-items:center;justify-content:center;padding:28px}
    .image-zoom-overlay.open{display:flex}
    .image-zoom-dialog{position:relative;width:min(96vw,1400px);max-height:94vh;display:flex;flex-direction:column;align-items:center;gap:10px}
    .image-zoom-dialog img{max-width:100%;max-height:82vh;width:auto;height:auto;object-fit:contain;background:#fff;border-radius:12px;box-shadow:0 18px 60px rgba(0,0,0,.5)}
    .image-zoom-caption{max-width:1000px;color:#fff;text-align:center;font-size:.9rem;margin:0}
    .image-zoom-close{position:absolute;top:-16px;right:-6px;width:44px;height:44px;border:0;border-radius:50%;background:#fff;color:#10243b;font-size:1.7rem;line-height:1;font-weight:800;box-shadow:0 6px 18px rgba(0,0,0,.35);z-index:2}
    @media(max-width:700px){.image-zoom-overlay{padding:16px}.image-zoom-dialog img{max-height:78vh}.image-zoom-close{top:-10px;right:-2px}}
  `;
  document.head.appendChild(zoomStyle);

  const zoomOverlay=document.createElement('div');
  zoomOverlay.className='image-zoom-overlay';
  zoomOverlay.setAttribute('role','dialog');
  zoomOverlay.setAttribute('aria-modal','true');
  zoomOverlay.setAttribute('aria-label','Larger image view');
  zoomOverlay.innerHTML=`<div class="image-zoom-dialog"><button class="image-zoom-close" type="button" aria-label="Close larger image">×</button><img alt=""><p class="image-zoom-caption"></p></div>`;
  document.body.appendChild(zoomOverlay);
  const zoomImage=zoomOverlay.querySelector('img');
  const zoomCaption=zoomOverlay.querySelector('.image-zoom-caption');
  const zoomClose=zoomOverlay.querySelector('.image-zoom-close');
  let zoomReturnFocus=null;

  function prepareZoomImages(){
    panel.querySelectorAll('img').forEach(img=>{
      img.classList.add('zoomable-image');
      if(!img.hasAttribute('tabindex')) img.tabIndex=0;
      img.setAttribute('role','button');
      const label=img.alt?.trim() || 'science diagram';
      img.setAttribute('aria-label',`Open larger image: ${label}`);
      img.title='Click to enlarge';
    });
  }

  function openZoom(img){
    zoomReturnFocus=img;
    zoomImage.src=img.currentSrc || img.src;
    zoomImage.alt=img.alt || '';
    const figure=img.closest('figure');
    const caption=figure?.querySelector('figcaption')?.textContent?.trim() || img.alt?.trim() || '';
    zoomCaption.textContent=caption;
    zoomCaption.style.display=caption?'block':'none';
    zoomOverlay.classList.add('open');
    document.body.classList.add('image-zoom-open');
    zoomClose.focus();
  }

  function closeZoom(){
    zoomOverlay.classList.remove('open');
    document.body.classList.remove('image-zoom-open');
    zoomImage.removeAttribute('src');
    if(zoomReturnFocus?.isConnected) zoomReturnFocus.focus();
  }

  panel.addEventListener('click',e=>{
    const img=e.target.closest('img.zoomable-image');
    if(!img || !panel.contains(img)) return;
    e.preventDefault();
    e.stopPropagation();
    openZoom(img);
  });
  panel.addEventListener('keydown',e=>{
    const img=e.target.closest('img.zoomable-image');
    if(!img || !panel.contains(img) || (e.key!=='Enter' && e.key!==' ')) return;
    e.preventDefault();
    openZoom(img);
  });
  zoomClose.addEventListener('click',closeZoom);
  zoomOverlay.addEventListener('click',e=>{if(e.target===zoomOverlay) closeZoom();});
  document.addEventListener('keydown',e=>{if(e.key==='Escape' && zoomOverlay.classList.contains('open')) closeZoom();});
  new MutationObserver(prepareZoomImages).observe(panel,{childList:true,subtree:true});

  render();
  prepareZoomImages();
})();
